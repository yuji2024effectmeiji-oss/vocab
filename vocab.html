`; } document.getElementById("next-btn").style.display = "block"; document.getElementById("submit-btn").disabled = true; }); // ⭐ 次も必ずランダム抽選 document.getElementById("next-btn").addEventListener("click", pickRandomQuestion); // 初期表示 window.addEventListener("load", pickRandomQuestion);
